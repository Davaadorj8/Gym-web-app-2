    return members.filter((m) => {
      if (m.status !== 'Active' || !m.expirationDate) return false;
      try {
        const today = new Date();
        today.setHours(0, 0, 0, 0);
        const expDate = new Date(m.expirationDate);
        const diffTime = expDate.getTime() - today.getTime();
        const diffDays = Math.ceil(diffTime / 86400000);
        return diffDays >= 0 && diffDays <= 7;
      } catch {
        return false;
      }
    }).length;

    const sorted = [...hourlyTraffic].sort((a, b) => b.count - a.count);
    const peak = sorted[0];
    return peak && peak.count > 0 ? `${peak.hour} (${peak.count})` : 'N/A';

      inactive: 0,
    };
    list.forEach((loc) => {
      const st = dashboard.lockerStatuses[loc];
      if (st && st in counts) {
        counts[st as keyof typeof counts]++;
      }

    let expiringSoonCount = 0;

      if (st === 'expired') expiredCount++;
      else if (st === 'expiring_soon') expiringSoonCount++;
      else if (st === 'fresh') freshCount++;

      Snacks: { count: 0, stock: 0, totalValue: 0, salesRevenue: 0 },
      Vitamins: { count: 0, stock: 0, totalValue: 0, salesRevenue: 0 },
    };

      const cat = n.category || 'Supplements';
      if (!categoryMap[cat]) {

        n.category.toLowerCase().includes(q) ||
        (n.bestBeforeDate && n.bestBeforeDate.includes(q));

        matchesStatus = n.stock > 5;

        matchesStatus = n.stock === 0;

        matchesStatus = getNutrientExpiryStatus(n.bestBeforeDate) === 'expiring_soon';
